# دالة ذكية جداً لقراءة أي ملف إكسل بأي صيغة
# ============================================
def read_flexible_excel(file_source):
  """تحليل وقراءة أي ملف إكسل مرن والتعرف التلقائي على الأعمدة"""
  try:
    xls = pd.ExcelFile(file_source)

    # البحث عن أول شيت يحتوي على بيانات
    df = None
    for sheet in xls.sheet_names:
      try:
        temp_df = pd.read_excel(xls, sheet_name=sheet)
        # تجربة القراءة مع التجاوز عن الصفوف الفارغة في الأعلى
        if temp_df.dropna(how="all").shape[0] > 0:
          df = temp_df
          break
      except:
        continue

    if df is None or df.empty:
      return None

    # إذا كانت الهيدرات في صف متأخر (كما في بعض تقارير الإكسل)
    if "المشروع" not in df.columns and "Project" not in [
        str(c) for c in df.columns
    ]:
      for header_row in range(1, min(5, len(df))):
        try:
          temp_df = pd.read_excel(xls, header=header_row)
          if any(
              k in str(c).lower()
              for c in temp_df.columns
              for k in ["مشروع", "project", "حالة", "status"]
          ):
            df = temp_df
            break
        except:
          pass

    # تنظيف وتوحيد مسافات الأعمدة
    df.columns = [str(c).strip() for c in df.columns]

    # قاموس الذكاء لربط وتوحيد أسماء الأعمدة المختلفة (عربي / إنجليزي)
    column_mapping = {}
    for col in df.columns:
      c_lower = col.lower()
      if any(k in c_lower for k in ["مشروع", "project", "اسم"]):
        column_mapping[col] = "المشروع"
      elif any(k in c_lower for k in ["حالة", "status", "state"]):
        column_mapping[col] = "الحالة"
      elif any(k in c_lower for k in ["مدير", "manager", "lead"]):
        column_mapping[col] = "مدير_المشروع"
      elif any(k in c_lower for k in ["جهة", "عميل", "client", "sector"]):
        column_mapping[col] = "الجهة"
      elif any(
          k in c_lower
          for k in [
              "إجمال",
              "عقد",
              "contract",
              "total",
              "value",
              "قيمة",
              "ميزانية",
          ]
      ):
        column_mapping[col] = "القيمة_الاجمالية"
      elif any(
          k in c_lower
          for k in ["صرف", "مصروف", "spent", "cost", "actual", "مدفوع"]
      ):
        column_mapping[col] = "ما_تم_صرفه"

    df = df.rename(columns=column_mapping)

    # إضافة أعمدة حماية في حال عدم وجودها لمنع KeyError
    if "المشروع" not in df.columns:
      df["المشروع"] = [f"مشروع {i+1}" for i in range(len(df))]
    if "الحالة" not in df.columns:
      df["الحالة"] = "جاري"
    if "مدير_المشروع" not in df.columns:
      df["مدير_المشروع"] = "غير محدد"
    if "الجهة" not in df.columns:
      df["الجهة"] = "غير محدد"
    if "القيمة_الاجمالية" not in df.columns:
      df["القيمة_الاجمالية"] = 0
    if "ما_تم_صرفه" not in df.columns:
      df["ما_تم_صرفه"] = 0

    # تنظيف الحقول الأرقام
    df["القيمة_الاجمالية"] = pd.to_numeric(
        df["القيمة_الاجمالية"], errors="coerce"
    ).fillna(0)
    df["ما_تم_صرفه"] = pd.to_numeric(
        df["ما_تم_صرفه"], errors="coerce"
    ).fillna(0)
    df["المتبقي"] = df["القيمة_الاجمالية"] - df["ما_تم_صرفه"]

    return df

  except Exception as e:
    st.error(f"⚠️ خطأ في قراءة ملف الإكسل: {e}")
    return None
