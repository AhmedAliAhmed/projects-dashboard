# ============================================
# دالة ذكية لمطابقة تقرير مستخلصات المشاريع
# ============================================
def read_invoice_excel(file_source):
  """تحليل وقراءة تقرير مستخلصات المشاريع بأي صيغة إكسل"""
  try:
    xls = pd.ExcelFile(file_source)

    df = None
    for sheet in xls.sheet_names:
      try:
        temp_df = pd.read_excel(xls, sheet_name=sheet)
        if temp_df.dropna(how="all").shape[0] > 0:
          df = temp_df
          break
      except:
        continue

    if df is None or df.empty:
      return None

    # التعرف التلقائي على صف الهيدر (Header Row)
    cols_str = " ".join([str(c) for c in df.columns])
    if not any(
        k in cols_str for k in ["مشروع", "مستخلص", "مستحقة", "Project"]
    ):
      for r in range(1, min(6, len(df))):
        try:
          temp_df = pd.read_excel(xls, header=r)
          c_str = " ".join([str(c) for c in temp_df.columns])
          if any(k in c_str for k in ["مشروع", "مستخلص", "مستحقة", "Project"]):
            df = temp_df
            break
        except:
          pass

    df.columns = [str(c).strip() for c in df.columns]

    # قاموس الذكاء لمطابقة أعمدة تقرير المستخلصات
    col_map = {}
    for c in df.columns:
      cl = c.lower()
      if "مشروع" in cl or "project" in cl:
        col_map[c] = "المشروع"
      elif "مستحقة" in cl or "مستحق" in cl or "إجمالي" in cl or "اجمالي" in cl:
        col_map[c] = "المستحقة"
      elif "مرفوعة" in cl or "مرفوع" in cl:
        col_map[c] = "المرفوعة"
      elif (
          "أمر دفع" in cl
          or "امر دفع" in cl
          or "دفع" in cl
          or "صرف" in cl
          or "مسدد" in cl
      ):
        col_map[c] = "امر_دفع"
      elif "مستهدف" in cl or "هدف" in cl:
        col_map[c] = "مستهدف_رفعها"

    df = df.rename(columns=col_map)

    # استبعاد صفوف الإجماليات المكتوبة في أسفل الإكسل لتجنب تكرار الحسابات
    if "المشروع" in df.columns:
      df = df[
          ~df["المشروع"]
          .astype(str)
          .str.contains("إجمالي|اجمالي|مجموع|المستخلصات المرفوعة", na=False)
      ]
      df = df[df["المشروع"].notna()]

    # إنشاء الأعمدة إن لم توجد
    if "المشروع" not in df.columns:
      df["المشروع"] = [f"مشروع {i+1}" for i in range(len(df))]

    for num_col in ["المستحقة", "المرفوعة", "امر_دفع", "مستهدف_رفعها"]:
      if num_col not in df.columns:
        df[num_col] = 0
      else:
        df[num_col] = pd.to_numeric(df[num_col], errors="coerce").fillna(0)

    return df

  except Exception as e:
    st.error(f"⚠️ خطأ أثناء قراءة ملف المستخلصات: {e}")
    return None
